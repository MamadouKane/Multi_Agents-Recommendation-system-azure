--------------------------------------------------------------------------------
 1. UV: Python environments
--------------------------------------------------------------------------------

# Install a uv-managed Python version
uv python install 3.11
uv python list                                  # available / installed versions

# Project venv (Python 3.11)
uv venv --python 3.11 .venv
source .venv/bin/activate                       # activate (optional when using make)
deactivate                                      # deactivate

# Install the project in editable mode + extras
uv pip install --python .venv/bin/python -e ".[dev]"
uv pip install --python .venv/bin/python -e ".[dev,ml,evals]"

# Inspect the venv
uv pip list --python .venv
uv pip show --python .venv/bin/python <package>  # version actually installed

# Resolve dependencies WITHOUT installing (debug conflicts)
uv pip compile requirements.txt --python-version 3.12

# Isolated CLI tools (outside the project venv): az, etc.
uv tool list
uv tool install <tool>
uv tool uninstall <tool>
uv tool update-shell                            # adds ~/.local/bin to PATH

# ⚠️ Azure CLI via uv: --prerelease allow is REQUIRED
#   (otherwise uv silently falls back to azure-cli 2.0.67, from 2019)
uv tool install azure-cli --python 3.12 --prerelease allow
uv tool upgrade azure-cli --prerelease allow    # do NOT use `az upgrade`
#   The shipped `az` script calls whatever `python` is on PATH (Anaconda) → replace it
#   with this wrapper (redo it if uv recreates the symlink after an upgrade):
#   rm ~/.local/bin/az
#   cat > ~/.local/bin/az <<'EOF'
#   #!/usr/bin/env bash
#   exec "$HOME/.local/share/uv/tools/azure-cli/bin/python" -m azure.cli "$@"
#   EOF
#   chmod +x ~/.local/bin/az

 4. AZ: output formats and queries (JMESPath)
--------------------------------------------------------------------------------

-o table | json | tsv | yaml                     # tsv = best for capturing a value into a shell variable
--query "<JMESPath expression>"

# Examples
az group list --query "[].name" -o tsv
az resource list -g <rg> --query "[].{name:name, type:type}" -o table
RG_ID=$(az group show -n <rg> --query id -o tsv)

 11. AZ: Container Registry and Container Apps (day 6)
--------------------------------------------------------------------------------

az acr login -n <acr-name>
az acr repository list -n <acr-name> -o table
# Build in Azure (avoids the Mac arm64 issue):
az acr build -r <acr-name> -t coffee-ai-api:<tag> -f src/api/Dockerfile .
# Local build targeting Azure:
docker build --platform linux/amd64 -t coffee-ai-api:local -f src/api/Dockerfile .

az containerapp show -n <app> -g <rg> --query properties.configuration.ingress.fqdn -o tsv
az containerapp logs show -n <app> -g <rg> --follow
az containerapp revision list -n <app> -g <rg> -o table
# Blue/green: split traffic between revisions / rollback
az containerapp ingress traffic set -n <app> -g <rg> --revision-weight <revision>=100
